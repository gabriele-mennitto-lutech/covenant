import { Component, Input, OnChanges, OnDestroy } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { RouterLink } from '@angular/router';
import { BackendService, CrossReference, CrossSourceResult, CrossSourceType } from '../../services/backend.service';

/** Pannello integrabile in schede e popup, con drill-down e paginazione lato server. */
@Component({
  selector: 'app-cross-source',
  standalone: true,
  imports: [CommonModule, FormsModule, RouterLink],
  templateUrl: './cross-source.component.html',
  styleUrls: ['./cross-source.component.css']
})
export class CrossSourceComponent implements OnChanges, OnDestroy {
  @Input({ required: true }) sourceType!: CrossSourceType;
  @Input({ required: true }) sourceId!: number;
  @Input() revision: unknown;

  readonly categories: { type: CrossSourceType; label: string }[] = [
    { type: 'hosts', label: 'Host' }, { type: 'services', label: 'Servizi' },
    { type: 'sites', label: 'Sites' }, { type: 'cves', label: 'CVE' }
  ];
  result: CrossSourceResult | null = null;
  target: CrossSourceType | '' = '';
  search = '';
  page = 1;
  loading = false;
  error: string | null = null;
  collapsed = false;
  currentType!: CrossSourceType;
  currentId!: number;
  private request?: AbortController;
  private timer?: ReturnType<typeof setTimeout>;

  constructor(private readonly backend: BackendService) {}

  ngOnChanges(): void {
    this.reset();
  }

  ngOnDestroy(): void {
    this.request?.abort();
    clearTimeout(this.timer);
  }

  reset(): void {
    this.explore({ type: this.sourceType, id: this.sourceId });
  }

  toggleCollapsed(): void {
    this.collapsed = !this.collapsed;
  }

  explore(reference: Pick<CrossReference, 'type' | 'id'>): void {
    clearTimeout(this.timer);
    this.currentType = reference.type;
    this.currentId = reference.id;
    this.target = '';
    this.search = '';
    this.page = 1;
    this.result = null;
    this.load();
  }

  filter(target: CrossSourceType | '' = this.target): void {
    clearTimeout(this.timer);
    this.target = target;
    this.page = 1;
    this.load();
  }

  searchChanged(): void {
    clearTimeout(this.timer);
    this.timer = setTimeout(() => this.filter(), 300);
  }

  async load(page = this.page): Promise<void> {
    this.request?.abort();
    const request = new AbortController();
    this.request = request;
    this.loading = true;
    this.error = null;
    this.page = page;
    try {
      const result = await this.backend.getCrossReferences(this.currentType, this.currentId,
        { target: this.target || undefined, search: this.search || undefined, page }, request.signal);
      if (!request.signal.aborted) this.result = result;
    } catch (error) {
      if (!request.signal.aborted) {
        console.error('Failed to load cross-source references', error);
        this.error = 'Impossibile caricare i riferimenti. Riprova.';
        this.result = null;
      }
    } finally {
      if (!request.signal.aborted) this.loading = false;
    }
  }

  label(type: CrossSourceType): string {
    return this.categories.find(category => category.type === type)!.label;
  }

  route(reference: CrossReference): (string | number)[] {
    return reference.type === 'hosts' ? ['/machine', reference.id]
      : [reference.type === 'cves' ? '/cve-analysis' : `/${reference.type}`];
  }

  query(reference: CrossReference): Record<string, number> {
    if (reference.type === 'hosts') return {};
    const key = reference.type === 'sites' ? 'siteId' : reference.type === 'services' ? 'serviceId' : 'cveId';
    return { [key]: reference.id };
  }
}

<section class="cross-source" aria-label="Riferimenti cross source">
  <header>
    <button type="button" class="title-toggle" (click)="toggleCollapsed()"
      [attr.aria-expanded]="!collapsed" aria-controls="cross-source-content">
      <h3>Riferimenti cross source</h3>
      <span class="chevron" [class.collapsed]="collapsed" aria-hidden="true">⌄</span>
    </button>
    <button type="button" (click)="load()" [disabled]="loading">Aggiorna</button>
  </header>
  <div *ngIf="!collapsed" id="cross-source-content">
    <p class="hint">Solo dati salvati. Le corrispondenze potenziali, incluse le CVE, richiedono verifica:
      non dimostrano vulnerabilità né presenza di una versione software. Sono utilizzati solo dati persistiti.</p>
    <div *ngIf="currentType !== sourceType || currentId !== sourceId" class="exploring">
      <button type="button" (click)="reset()">Torna alla sorgente</button>
      <span *ngIf="result">{{ label(currentType) }}: {{ result.source.label }}</span>
    </div>
    <div class="filters">
      <button type="button" [class.active]="!target" (click)="filter('')" [disabled]="loading">Tutti</button>
      <button *ngFor="let category of categories" type="button" [class.active]="target === category.type"
        (click)="filter(category.type)" [disabled]="loading">
        {{ category.label }} <span *ngIf="result">({{ result.counts[category.type] }})</span>
      </button>
      <input type="search" aria-label="Cerca nei riferimenti" placeholder="Cerca nei riferimenti..."
        maxlength="200" [(ngModel)]="search" (ngModelChange)="searchChanged()" />
    </div>
    <p *ngIf="loading" role="status">Caricamento riferimenti...</p>
    <div *ngIf="error" role="alert" class="error">{{ error }} <button type="button" (click)="load()">Riprova</button></div>
    <ng-container *ngIf="!loading && result">
      <p *ngIf="!result.items.length" class="hint">Nessuna corrispondenza nei dati disponibili.
        L'assenza di riferimenti non esclude un impatto.</p>
      <article *ngFor="let item of result.items">
        <div class="reference-heading">
          <strong>{{ label(item.type) }} · {{ item.label }}</strong>
          <span class="badge" [class.direct]="item.confidence === 'direct'">
            {{ item.confidence === 'direct' ? 'Riferimento diretto' : 'Potenziale / da verificare' }}
          </span>
        </div>
        <p *ngIf="item.context" class="hint">{{ item.context }}</p>
        <ul><li *ngFor="let reason of item.reasons">{{ reason }}</li></ul>
        <div class="actions">
          <a [routerLink]="route(item)" [queryParams]="query(item)">Apri scheda</a>
          <button type="button" (click)="explore(item)">Esplora collegamenti</button>
        </div>
      </article>
      <footer *ngIf="result.total > 0">
        <span>{{ result.total }} riferimenti · {{ result.page }} / {{ result.totalPages }}</span>
        <button type="button" [disabled]="page <= 1" (click)="load(page - 1)">Precedenti</button>
        <button type="button" [disabled]="page >= result.totalPages" (click)="load(page + 1)">Successivi</button>
      </footer>
    </ng-container>
  </div>
</section>
